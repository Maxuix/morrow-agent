import re


class PhoneNumber:
    def __init__(self, number):
        cleaned = re.sub(r"[().+\-\s]", "", number)
        if not cleaned.isdigit():
            raise ValueError("letters or punctuation not permitted")
        if len(cleaned) == 11:
            if not cleaned.startswith("1"):
                raise ValueError("11 digits must start with 1")
            cleaned = cleaned[1:]
        if len(cleaned) < 10:
            raise ValueError("must not be fewer than 10 digits")
        if len(cleaned) > 10:
            raise ValueError("must not be greater than 11 digits")
        if cleaned[0] in "01":
            raise ValueError("area code must start with 2 through 9")
        if cleaned[3] in "01":
            raise ValueError("exchange code must start with 2 through 9")
        self.number = cleaned
        self.area_code = cleaned[:3]

    def pretty(self):
        return f"({self.number[:3]})-{self.number[3:6]}-{self.number[6:]}"
