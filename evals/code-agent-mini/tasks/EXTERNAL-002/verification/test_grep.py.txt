from pathlib import Path

from grep import grep


def write_files(root: Path):
    first = root / "first.txt"
    second = root / "second.txt"
    first.write_text("Alpha\nbeta\nAlpha beta\n", encoding="utf-8")
    second.write_text("beta\nALPHA\nomega\n", encoding="utf-8")
    return str(first), str(second)


def test_single_file_and_line_numbers(tmp_path):
    first, _second = write_files(tmp_path)
    assert grep("Alpha", "", [first]) == "Alpha\nAlpha beta\n"
    assert grep("beta", "-n", [first]) == "2:beta\n3:Alpha beta\n"


def test_case_exact_and_inverse_flags(tmp_path):
    first, _second = write_files(tmp_path)
    assert grep("alpha", "-i -x", [first]) == "Alpha\n"
    assert grep("beta", "-v", [first]) == "Alpha\n"


def test_multiple_files_are_prefixed_in_order(tmp_path):
    first, second = write_files(tmp_path)
    assert grep("beta", "", [first, second]) == (
        f"{first}:beta\n{first}:Alpha beta\n{second}:beta\n"
    )


def test_file_names_are_emitted_once(tmp_path):
    first, second = write_files(tmp_path)
    assert grep("alpha", "-i -l", [first, second]) == f"{first}\n{second}\n"
